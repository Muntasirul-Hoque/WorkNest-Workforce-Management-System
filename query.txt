CREATE TABLE `User` (
    `UserName` VARCHAR(50) NOT NULL,
    `Password` VARCHAR(255) NOT NULL,
    `Role` VARCHAR(50) NOT NULL,
    `Dept` VARCHAR(50) NOT NULL,
    `Salary` DECIMAL(10,2) NOT NULL,
    `FirstName` VARCHAR(50) NOT NULL,
    `LastName` VARCHAR(50) NOT NULL,
    `DOB` DATE NOT NULL,
    `Gender` VARCHAR(10) NOT NULL,
    `Email` VARCHAR(100) NOT NULL,
    `Phone` VARCHAR(20) NOT NULL,
    `Address` VARCHAR(255) NOT NULL,
    `JoinDate` DATE NOT NULL,
    `Status` VARCHAR(20) DEFAULT 'Active',
    PRIMARY KEY (`UserName`)
);

CREATE TABLE `hr` (
    `HRID` INT AUTO_INCREMENT PRIMARY KEY,
    `UserName` VARCHAR(50) NOT NULL,
    FOREIGN KEY (`UserName`) REFERENCES `User`(`UserName`) ON DELETE CASCADE
);

CREATE TABLE `Employee` (
    `EmployeeID` INT AUTO_INCREMENT PRIMARY KEY,
    `UserName` VARCHAR(50) NOT NULL,
    `ManagerID` INT NULL,
    FOREIGN KEY (`UserName`) REFERENCES `User`(`UserName`) ON DELETE CASCADE
);

CREATE TABLE `Manager` (
    `ManagerID` INT AUTO_INCREMENT PRIMARY KEY,
    `UserName` VARCHAR(50) NOT NULL,
    `EmployeeID` INT NULL,
    FOREIGN KEY (`UserName`) REFERENCES `User`(`UserName`) ON DELETE CASCADE
);

CREATE TABLE `Leave` (
    `LeaveID` INT AUTO_INCREMENT PRIMARY KEY,
    `UserName` VARCHAR(50) NOT NULL,
    `SD` DATE NOT NULL,
    `ED` DATE NOT NULL,
    `Reason` TEXT NOT NULL,
    `Status` VARCHAR(20) DEFAULT 'Pending',
    FOREIGN KEY (`UserName`) REFERENCES `User`(`UserName`) ON DELETE CASCADE
);

CREATE TABLE `Promotion` (
    `PromotionID` INT AUTO_INCREMENT PRIMARY KEY,
    `EmployeeID` INT NOT NULL,
    `PromotionDate` DATE NOT NULL,
    `Status` VARCHAR(20) DEFAULT 'Pending',
    FOREIGN KEY (`EmployeeID`) REFERENCES `Employee`(`EmployeeID`) ON DELETE CASCADE
);

-- NEW TABLE NEEDED FOR MANAGER ASSIGNING TASKS:
CREATE TABLE `Task` (
    `TaskID` INT AUTO_INCREMENT PRIMARY KEY,
    `ManagerID` INT NOT NULL,
    `EmployeeID` INT NOT NULL,
    `Title` VARCHAR(100) NOT NULL,
    `Description` TEXT NOT NULL,
    `T_Status` VARCHAR(20) DEFAULT 'Pending',
    `AssignedDate` DATE NOT NULL,
    `DueDate` DATE NOT NULL,
    FOREIGN KEY (`ManagerID`) REFERENCES `Manager`(`ManagerID`) ON DELETE CASCADE,
    FOREIGN KEY (`EmployeeID`) REFERENCES `Employee`(`EmployeeID`) ON DELETE CASCADE
);
